\newcommand{\AssertSameWidth}[3]{%
  \sbox{\EdgeA}{#2}%
  \sbox{\EdgeB}{#3}%
  \ifdim\dimexpr\wd\EdgeA-\wd\EdgeB\relax<0pt
    \dimen0=\dimexpr\wd\EdgeB-\wd\EdgeA\relax
  \else
    \dimen0=\dimexpr\wd\EdgeA-\wd\EdgeB\relax
  \fi
  \ifdim\dimen0<0.01pt
    \typeout{GCK-EDGE PASS #1: \the\wd\EdgeA = \the\wd\EdgeB}%
  \else
    \PackageError{gckanbun-edge-test}{Width mismatch in #1:
      \the\wd\EdgeA\space vs \the\wd\EdgeB}{The tested boxes should have equal widths.}%
  \fi
}
\newcommand{\AssertSameMetrics}[3]{%
  \sbox{\EdgeA}{#2}%
  \sbox{\EdgeB}{#3}%
  \ifdim\wd\EdgeA=\wd\EdgeB
    \ifdim\ht\EdgeA=\ht\EdgeB
      \ifdim\dp\EdgeA=\dp\EdgeB
        \typeout{GCK-EDGE PASS #1: identical box metrics}%
      \else
        \PackageError{gckanbun-edge-test}{Depth mismatch in #1}{The tested boxes should have equal metrics.}%
      \fi
    \else
      \PackageError{gckanbun-edge-test}{Height mismatch in #1}{The tested boxes should have equal metrics.}%
    \fi
  \else
    \PackageError{gckanbun-edge-test}{Width mismatch in #1}{The tested boxes should have equal metrics.}%
  \fi
}
